<script>
$(document).ready(function() {
    let html5QrCode = null;
    let isProcessing = false;
    let lastScannedData = null;
    const bModal = new bootstrap.Modal('#scannerModal');

    function initSelect2(ctx = document) {
        $(ctx).find('select[name$="-drug"]').select2({
            theme: 'bootstrap-5', width: '100%', dir: 'rtl',
            ajax: { url: '/drugs/api/drugs/search/', dataType: 'json' }
        });
    }

    window.updateNumbers = function() {
        let grand = 0;
        $('.item-row-card').each(function(i) {
            $(this).find('.num').text(i + 1);
            let q = parseFloat($(this).find('.qty-input').val()) || 0;
            let p = parseFloat($(this).find('.unit-price-hidden').val()) || 0;
            grand += (q * p);
        });
        $('#grand-total-display').text(grand.toLocaleString('fa-IR'));
        $('#id_items-TOTAL_FORMS').val($('.item-row-card').length);
    }

    // منطق اسکن موفق
    function onScanSuccess(text) {
        if (isProcessing) return;
        isProcessing = true;

        let gtin = text.startsWith('01') ? text.substring(2, 16) : text;
        let expiry = "";
        let expMatch = text.match(/17(\d{6})/);
        if (expMatch) expiry = `20${expMatch[1].substring(0,2)}-${expMatch[1].substring(2,4)}-${expMatch[1].substring(4,6)}`;

        // چک کردن تکراری
        let isDuplicate = false;
        $('.gtin-input').each(function() { if($(this).val() === gtin) isDuplicate = true; });
        if(isDuplicate && !confirm("این دارو قبلاً اضافه شده. دوباره اضافه شود؟")) {
            isProcessing = false; return;
        }

        $.get('/drugs/api/find-by-barcode/', { barcode: gtin }, function(res) {
            if (res.status === 'found') {
                html5QrCode.pause(true);
                lastScannedData = { gtin, expiry, id: res.drug_id, name: res.drug_name };
                $('#scanned-drug-name').text(res.drug_name);
                $('#quantity-prompt').removeClass('d-none');
                $('#quick-qty-input').focus().select();
            } else {
                alert("بارکد یافت نشد!");
                isProcessing = false;
            }
        }).fail(() => { alert("خطا در سرور"); isProcessing = false; });
    }

    // استارت اسکنر
    async function startCam(id) {
        try {
            if (html5QrCode && html5QrCode.getState() === 2) await html5QrCode.stop();
            await html5QrCode.start(id, { fps: 10, qrbox: 250 }, onScanSuccess);
        } catch (e) { console.error(e); }
    }

    $('#main-scanner-trigger').click(() => bModal.show());

    document.getElementById('scannerModal').addEventListener('shown.bs.modal', async () => {
        if (!html5QrCode) html5QrCode = new Html5Qrcode("qr-reader");
        const devices = await Html5Qrcode.getCameras();
        if (devices.length) {
            const $sel = $('#camera-select').empty();
            devices.forEach(d => $sel.append(`<option value="${d.id}">${d.label}</option>`));
            let back = devices.find(d => d.label.toLowerCase().includes('back'));
            let targetId = back ? back.id : devices[0].id;
            $sel.val(targetId);
            startCam(targetId);
        }
    });

    document.getElementById('scannerModal').addEventListener('hidden.bs.modal', async () => {
        if (html5QrCode && html5QrCode.getState() !== 1) await html5QrCode.stop();
        isProcessing = false;
    });

    $('#camera-select').change(function() { startCam($(this).val()); });

    $('#confirm-qty-btn').click(function() {
        const qty = $('#quick-qty-input').val();
        const count = $('.item-row-card').length;
        const row = $($('#empty-row-tmpl').html().replace(/__prefix__/g, count));
        
        row.find('.gtin-input').val(lastScannedData.gtin);
        row.find('.expiry-input').val(lastScannedData.expiry);
        row.find('.qty-input').val(qty);
        row.find('select').append(new Option(lastScannedData.name, lastScannedData.id, true, true));
        
        $('#formset-container').prepend(row);
        initSelect2(row); updateNumbers();
        
        $('#quantity-prompt').addClass('d-none');
        isProcessing = false;
        html5QrCode.resume();
    });

    $(document).on('input', '.currency-input', function() {
        let v = $(this).val().replace(/[^\d]/g, '');
        $(this).val(v.replace(/\B(?=(\d{3})+(?!\d))/g, ","));
        $(this).closest('.input-group').find('.unit-price-hidden').val(v);
        updateNumbers();
    });

    initSelect2(); updateNumbers();
});
</script>
